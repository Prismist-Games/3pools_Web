# Fate Wall (命运网格) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add the Fate Wall system to the v2 turn-based prototype — a 4×4 player-side grid where charms are placed (via ✨ cells) and drawn from (Luck draw each turn + Doom draw replacing the old danger grid).

**Architecture:** New `fateWall` state lives in `useGameLogic`. Three new modal components handle ✨ placement, Luck draw, and Doom draw (each modal blocks the turn flow). The old 10-cell danger grid is removed and its doom resolution replaced by the fate wall row/col draw system.

**Tech Stack:** React 18, Vite, Tailwind CSS 3 — follows existing patterns in the codebase.

- All game state lives in `useGameLogic`; `GameCore` is pure rendering + wiring.
- No test suite — verification is always `npm run dev` + manual browser check.
- Use `Math.random().toString(36).substr(2,9) + Date.now().toString(36)` for UIDs (never `crypto.randomUUID()`).

```js
/** Weighted random pick from CHARM_GENERATION_WEIGHTS */
export function rollCharmType() {
  const entries = Object.entries(CHARM_GENERATION_WEIGHTS);
  const total = entries.reduce((s, [, w]) => s + w, 0);
  let roll = Math.random() * total;
  for (const [type, weight] of entries) {
    roll -= weight;
    if (roll <= 0) return type;
  }
  return entries[entries.length - 1][0];
}

  return {
    id: generateUID(),
    type,
    isPersistent: CHARM_CONFIGS[type].isPersistent,
    isPassive: CHARM_CONFIGS[type].isPassive,
    isBlank: CHARM_CONFIGS[type].isBlank ?? false,
    growthCount: 0,
    usesLeft: type === CHARM_TYPES.ALCHEMY_POT ? 3 : undefined,
  };
}
```

```js
// src/utils/fateWallHelpers.js
// Pure helper functions for the 4×4 fate wall grid.

import { CHARM_TYPES } from '../data/charms';

/** Convert flat index to { row, col } */
export function indexToRC(index) {
  return { row: Math.floor(index / 4), col: index % 4 };
}

/** Convert { row, col } to flat index */
export function rcToIndex(row, col) {
  return row * 4 + col;
}

/** Get flat indices of 4-directional neighbors (no diagonals). Out-of-bounds excluded. */
export function getNeighbors(index) {
  const { row, col } = indexToRC(index);
  const neighbors = [];
  if (row > 0) neighbors.push(rcToIndex(row - 1, col));
  if (row < 3) neighbors.push(rcToIndex(row + 1, col));
  if (col > 0) neighbors.push(rcToIndex(row, col - 1));
  if (col < 3) neighbors.push(rcToIndex(row, col + 1));
  return neighbors;
}

/** Get all 4 flat indices for a row (0–3). */
export function getRowIndices(row) {
  return [0, 1, 2, 3].map(col => rcToIndex(row, col));
}

/** Get all 4 flat indices for a col (0–3). */
export function getColIndices(col) {
  return [0, 1, 2, 3].map(row => rcToIndex(row, col));
}

/**
 * Determine the doom hit target index for one Doom draw.
 *
 * Rules (in priority order):
 * 1. Compute the 4 candidate indices for the given row or col.
 * 2. GUARD_STONE protection: for each GUARD_STONE in the row/col, exclude
 *    its 4-directional neighbors from the candidate list. The guard stone
 *    itself remains a valid candidate (it protects others, not itself).
 * 3. BAIT priority: if any remaining candidate is a BAIT charm, doom hits
 *    a random bait among them (ignoring non-bait candidates).
 * 4. Otherwise: uniformly random among all remaining candidates (including
 *    null/empty cells).
 *
 * @param {Array} cells - flat 4×4 cells array (index 0–15, null = empty)
 * @param {'row'|'col'} direction
 * @param {number} lineIndex - 0–3 which row or col
 * @returns {number} index of the cell that doom hits
 */
export function getDoomTarget(cells, direction, lineIndex) {
  const indices = direction === 'row'
    ? getRowIndices(lineIndex)
    : getColIndices(lineIndex);

  // Step 1: Start with all 4 indices
  let candidates = [...indices];

  // Step 2: Guard stone protection — find guard stones in this line
  const guardStoneIndices = indices.filter(
    i => cells[i]?.type === CHARM_TYPES.GUARD_STONE
  );
  for (const gsIdx of guardStoneIndices) {
    const protected_ = getNeighbors(gsIdx);
    // Remove protected indices from candidates (guard stone itself stays)
    candidates = candidates.filter(
      i => i === gsIdx || !protected_.includes(i)
    );
  }

  // Step 3: Bait priority
  const baitCandidates = candidates.filter(
    i => cells[i]?.type === CHARM_TYPES.BAIT
  );
  if (baitCandidates.length > 0) {
    return baitCandidates[Math.floor(Math.random() * baitCandidates.length)];
  }

  // Step 4: Uniform random among remaining candidates
  return candidates[Math.floor(Math.random() * candidates.length)];
}

/**
 * Check whether ALL candidates in a line are excluded by guard stones
 * (meaning the player must re-pick a different line).
 */
export function isLineFullyProtected(cells, direction, lineIndex) {
  const indices = direction === 'row'
    ? getRowIndices(lineIndex)
    : getColIndices(lineIndex);

  let candidates = [...indices];
  const guardStoneIndices = indices.filter(
    i => cells[i]?.type === CHARM_TYPES.GUARD_STONE
  );
  for (const gsIdx of guardStoneIndices) {
    const protected_ = getNeighbors(gsIdx);
    candidates = candidates.filter(
      i => i === gsIdx || !protected_.includes(i)
    );
  }
  // If no candidates remain after exclusion, line is fully protected
  return candidates.length === 0;
}

/**
 * Check whether a line has at least one filled charm
 * (needed to validate Luck draw selection — empty lines are invalid).
 */
export function lineHasCharm(cells, direction, lineIndex) {
  const indices = direction === 'row'
    ? getRowIndices(lineIndex)
    : getColIndices(lineIndex);
  return indices.some(i => cells[i] !== null);
}
```

```jsx
import React from 'react';
import { useLanguage } from '../../contexts/LanguageContext';
import { CHARM_CONFIGS, CHARM_TYPES } from '../../data/charms';

export function FateWallPlacementModal({ pendingCharm, fateWallCells, onPlace }) {
  const { t } = useLanguage();
  const charm = pendingCharm.charm;
  const cfg = CHARM_CONFIGS[charm.type];

  const canPlace = (index) => {
    const cell = fateWallCells[index];
    if (cell === null) return true;          // empty → ok
    if (cell.isBlank) return false;         // BLANK → forbidden
    return true;                             // occupied non-blank → overwrite ok
  };

  return (
    <div className="fixed inset-0 bg-black/60 flex items-center justify-center z-50">
      <div className="bg-gray-900 border border-yellow-500 rounded-xl p-6 max-w-sm w-full shadow-2xl">
        <h2 className="text-yellow-400 text-lg font-bold mb-2">{t('放置幸运符')}</h2>
        <div className="flex items-center gap-3 mb-4 p-3 bg-gray-800 rounded-lg">
          <span className="text-2xl">{cfg.icon}</span>
```

```js
    if (!copyMirrorState || copyMirrorState.step !== 'select_source') return;
    const emptySlots = fateWall.cells
        .map((c, i) => (c === null ? i : -1))
        .filter(i => i !== -1);
    if (emptySlots.length === 0) {
        setCopyMirrorState(null);
        return;
    }
    const sourceCharm = fateWall.cells[sourceIndex];
    const newCharm = generateCharm(sourceCharm.type); // fresh growthCount = 0
    setCopyMirrorState({
        step: 'select_target',
        charmIndex: copyMirrorState.charmIndex,
        sourceIndex,
        emptySlots,
        newCharm,
    });
};

const handleCopyMirrorSelectTarget = (targetIndex) => {
    if (!copyMirrorState || copyMirrorState.step !== 'select_target') return;
    placeCharm(targetIndex, copyMirrorState.newCharm);
    setCopyMirrorState(null);
    // Luck phase can now continue
    setLuckResult(prev => ({
        ...prev,
        effectDescription: `${t('复制')}: ${CHARM_CONFIGS[copyMirrorState.newCharm.type].name}`,
    }));
};
```
